cmake_minimum_required(VERSION 3.22.1)
project(video2text LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
# Android's x86_64 ABI guarantees SSE4.2, but does not guarantee AVX/FMA.
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/whisper)
# Inference must be optimized even in debug APKs; retain their normal debug symbols.
target_compile_options(whisper PRIVATE -O3)
target_compile_options(ggml-cpu PRIVATE -O3)
target_compile_options(ggml-base PRIVATE -O3)
add_library(transcription SHARED transcription.cpp)
target_link_libraries(transcription PRIVATE whisper)
target_link_options(transcription PRIVATE "-Wl,-z,max-page-size=16384")
